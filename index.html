        function updateSimulationTarget(angry, happy, neutral, other) {

        }

        // 파티클 (용암/화산재) 클래스

        class Particle {

            constructor(x, y, power) {

                this.x = x;

                this.y = y;

                this.vx = (Math.random() - 0.5) * power * 0.5; // X축 퍼짐

                this.vy = (Math.random() * -1 - 0.5) * power; // Y축 위로 분출

                this.size = Math.random() * 6 + 2;

                this.life = 1.0; // 수명

                this.decay = Math.random() * 0.02 + 0.01;

                // 용암 색상

                const colors = ['#ef4444', '#f97316', '#fbbf24', '#475569']; // Red, Orange, Yellow, Ash(Gray)

                this.color = colors[Math.floor(Math.random() * (power > 5 ? 4 : 3))]; 

            }

            update() {

                this.x += this.vx;

                this.vy += 0.1; // 중력

                this.y += this.vy;

                this.life -= this.decay;

                if(this.size > 0.1) this.size -= 0.05;

            }

            draw(ctx) {

                ctx.globalAlpha = this.life > 0 ? this.life : 0;

                ctx.fillStyle = this.color;

                ctx.beginPath();

                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);

                ctx.fill();

                ctx.globalAlpha = 1.0;

            }

        }


        // 애니메이션 루프

        function animate() {

            // 부드러운 압력 변화 (Easing)

            currentPressure += (targetPressure - currentPressure) * 0.05;

            

            // 게이지바 업데이트

            pressureGauge.style.height = `${Math.max(5, currentPressure)}%`;

            pressureText.innerText = `${Math.round(currentPressure * 100)} hPa`; // 임의의 단위 hPa

            

            // 압력에 따른 게이지 색상 변경

            if (currentPressure > 80) {

                pressureGauge.className = "gauge-bar w-10 rounded-full relative z-0 mb-2 bg-gradient-to-t from-red-600 to-red-400 shadow-[0_0_20px_rgba(239,68,68,0.8)]";

            } else if (currentPressure > 50) {

                pressureGauge.className = "gauge-bar w-10 rounded-full relative z-0 mb-2 bg-gradient-to-t from-orange-600 to-yellow-500 shadow-[0_0_15px_rgba(245,158,11,0.6)]";

            } else {

                pressureGauge.className = "gauge-bar w-10 rounded-full relative z-0 mb-2 bg-gradient-to-t from-slate-600 to-blue-500 shadow-none";

            }


            // 메시지 업데이트 로직

            updateEducationalMessage(currentPressure);


            // 화면 지우기

            ctx.clearRect(0, 0, canvas.width, canvas.height);


            const cx = canvas.width / 2;

            const cy = canvas.height;

            const volcanoHeight = canvas.height * 0.6;

            const craterY = cy - volcanoHeight;

            const craterWidth = 60;

            

            // 1. 배경 마그마 챔버 (지하) 그리기

            const magmaChamberY = cy - 20;

            const magmaLevel = currentPressure / 100; // 0.0 ~ 1.0

            

            ctx.fillStyle = '#1e293b'; // 지하 암석 색상

            ctx.fillRect(0, cy - 80, canvas.width, 80);


            // 챔버 내 마그마 (압력에 따라 크기와 붉은빛 변화)

            const chamberRadius = 50 + (magmaLevel * 20);

            const gradientChamber = ctx.createRadialGradient(cx, magmaChamberY, 10, cx, magmaChamberY, chamberRadius);

            gradientChamber.addColorStop(0, '#fef08a'); // 중심 뜨거움

            gradientChamber.addColorStop(0.5, '#ef4444');

            gradientChamber.addColorStop(1, 'transparent');

            

            ctx.fillStyle = gradientChamber;

            ctx.beginPath();

            ctx.ellipse(cx, magmaChamberY, chamberRadius * 1.5, chamberRadius, 0, 0, Math.PI * 2);

            ctx.fill();


            // 2. 화산(산) 형태 그리기

            ctx.fillStyle = '#334155'; // 화산 외부 색상

            ctx.beginPath();

            ctx.moveTo(0, cy - 80);

            ctx.lineTo(cx - craterWidth - 100, cy - 80);

            ctx.lineTo(cx - craterWidth/2, craterY); // 분화구 왼쪽

            ctx.lineTo(cx + craterWidth/2, craterWidth/2 > 0 ? craterY : craterY); // 분화구 오른쪽

            ctx.lineTo(cx + craterWidth + 100, cy - 80);

            ctx.lineTo(canvas.width, cy - 80);

            ctx.fill();


            // 화산 테두리 및 그림자 (입체감)

            ctx.strokeStyle = '#0f172a';

            ctx.lineWidth = 3;

            ctx.stroke();

            

            ctx.fillStyle = '#1e293b'; // 어두운 면

            ctx.beginPath();

            ctx.moveTo(cx, craterY);

            ctx.lineTo(cx + craterWidth/2, craterY);

            ctx.lineTo(cx + craterWidth + 100, cy - 80);

            ctx.lineTo(cx, cy - 80);

            ctx.fill();


            // 3. 화도 (Magma Conduit) - 압력에 따라 용암이 차오름

            const conduitWidth = 20 + (magmaLevel * 10);

            const lavaHeightY = magmaChamberY - (magmaChamberY - craterY) * magmaLevel;

            

            // 화도 배경 (빈 관)

            ctx.fillStyle = '#0f172a';

            ctx.fillRect(cx - conduitWidth/2, craterY, conduitWidth, magmaChamberY - craterY);


            // 차오르는 마그마

            const gradientConduit = ctx.createLinearGradient(cx, magmaChamberY, cx, lavaHeightY);

            gradientConduit.addColorStop(0, '#ef4444');

            gradientConduit.addColorStop(1, '#fbbf24');

            

            ctx.fillStyle = gradientConduit;

            ctx.fillRect(cx - conduitWidth/2 + 2, lavaHeightY, conduitWidth - 4, magmaChamberY - lavaHeightY);


            // 4. 폭발 (Particles)

            if (currentPressure > 80) { // 분노 인식 시

                // 파티클 생성

                for(let i=0; i<3; i++) {

                    particles.push(new Particle(cx + (Math.random()*20 - 10), craterY, 15));

                }

                isErupting = true;

            } else if (currentPressure > 50) { // 기쁨 등 에너지 증가 시

                if (Math.random() > 0.5) {

                    particles.push(new Particle(cx + (Math.random()*10 - 5), lavaHeightY, 5));

                }

                isErupting = false;

            } else {

                isErupting = false;

            }


            // 파티클 업데이트 및 그리기

            for (let i = particles.length - 1; i >= 0; i--) {

                let p = particles[i];

                p.update();

                p.draw(ctx);

                if (p.life <= 0) {

                    particles.splice(i, 1);

                }

            }


            requestAnimationFrame(animate);

        }


        function updateEducationalMessage(pressure) {

            const eduBlock = document.getElementById('eduMessageBlock');

            

            if (pressure > 80) {

                eduMessage.innerHTML = '현재 상태: <span class="text-red-500 font-black animate-pulse">화산 폭발 (분노 에너지 감지)</span>';

                eduDetail.innerHTML = '강한 감정(분노)이 높은 열에너지로 치환되었습니다!<br>마그마 챔버의 압력이 임계점을 넘어 화산재와 용암이 분출하는 <strong>폭발성 분화</strong>가 발생합니다.';

                eduBlock.style.borderColor = '#ef4444';

            } else if (pressure > 50) {

                eduMessage.innerHTML = '현재 상태: <span class="text-yellow-400 font-bold">마그마 상승 (에너지 증가)</span>';

                eduDetail.innerHTML = '감정의 변화(기쁨/흥분)로 지구 내부 온도가 상승했습니다.<br>마그마가 화도를 따라 상승하며 화산 가스가 분출되고 있습니다.';

                eduBlock.style.borderColor = '#eab308';

            } else {

                eduMessage.innerHTML = '현재 상태: <span class="text-blue-300 font-bold">안정기 (휴화산)</span>';

                eduDetail.innerHTML = '평온한 상태입니다. 내부 열에너지가 균형을 이루어<br>마그마 챔버의 압력이 낮게 유지되고 있습니다.';

                eduBlock.style.borderColor = '#475569';

            }

        }

            animate();
